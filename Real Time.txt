vec4 square(in vec2 fragCoord, in vec2 center , in float size)
{

    float xmin = round(center.x - size);
    float xmax = round(center.x + size);

    float ymin = round(center.y - size);
    float ymax = round(center.y + size);
    
    vec3 col  = vec3(1,1,1);
    
    if(fragCoord.x < xmax && fragCoord.x > xmin &&  fragCoord.y < ymax && fragCoord.y > ymin){

        col  = vec3(0,1,1);

    }else if (abs(fragCoord.x - xmax) < 1.0  && abs(fragCoord.x - xmax) > 0.0 && fragCoord.y < ymax && fragCoord.y > ymin){

        col  = vec3(0,0,0);

    }else if(abs(fragCoord.x - xmin) < 1.0  && abs(fragCoord.x - xmin) > 0.0 && fragCoord.y < ymax  && fragCoord.y > ymin){

        col  = vec3(0,0,0);
           
     }else if(abs(fragCoord.y - ymax) < 1.0  && abs(fragCoord.y - ymax) > 0.0 && fragCoord.x < xmax  && fragCoord.x > xmin){

        col  = vec3(0,0,0);

     }else if(abs(fragCoord.y - ymin) < 1.0  && abs(fragCoord.y - ymin) > 0.0 && fragCoord.x < xmax  && fragCoord.x > xmin){

        col  = vec3(0,0,0);

     }
        

       return vec4(col,1);
  
        
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{

    vec2 pos = iResolution.xy/2.0;
    float size = 100.0;
    
    fragColor = square(fragCoord ,pos, size);
    
   
}
